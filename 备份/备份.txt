// 收敛稳定性管理器
class ConvergenceStabilityManager {
private:
    deque<double> fitness_history;
    double historical_best_fitness;
    double* historical_best_solution;
    int dimension;
    int stagnation_count;
    int max_stagnation;
    double convergence_threshold;
    double improvement_threshold;
    bool convergence_detected;
    int consecutive_small_improvements;
    int max_consecutive_small_improvements;
    
public:
    ConvergenceStabilityManager(int dim, int max_stag = 5) : 
        dimension(dim), max_stagnation(max_stag), 
        historical_best_fitness(1e10), stagnation_count(0),
        convergence_threshold(1e-6), improvement_threshold(0.001),
        convergence_detected(false), consecutive_small_improvements(0),
        max_consecutive_small_improvements(3) {
        
        historical_best_solution = new double[dimension];
        for(int i = 0; i < dimension; i++) {
            historical_best_solution[i] = 0.0;
        }
    }
    
    ~ConvergenceStabilityManager() {
        delete[] historical_best_solution;
    }
    
    // 判断是否应该接受新的适应度值
    bool should_accept_fitness(double old_fitness, double new_fitness, double* new_solution, int iteration) {
        double improvement_rate = 0.0;
        if (abs(old_fitness) > 1e-8) {
            improvement_rate = (old_fitness - new_fitness) / abs(old_fitness);
        }
        
        // 更新历史记录
        fitness_history.push_back(new_fitness);
        if (fitness_history.size() > 10) {
            fitness_history.pop_front();
        }
        
        // 更新历史最佳
        if (new_fitness < historical_best_fitness) {
            historical_best_fitness = new_fitness;
            memcpy(historical_best_solution, new_solution, dimension * sizeof(double));
            stagnation_count = 0;
            consecutive_small_improvements = 0;
        } else {
            stagnation_count++;
        }
        
        // 检查连续小改进
        if (improvement_rate > 0 && improvement_rate < improvement_threshold) {
            consecutive_small_improvements++;
        } else if (improvement_rate >= improvement_threshold) {
            consecutive_small_improvements = 0;
        }
        
        // 收敛检测
        check_convergence();
        
        // === 改进的决策规则 ===

        // 1. 无条件接受正改进
        if (improvement_rate >= 0) {
            return true;
        }
        
        // 2. 严格拒绝大的负改进
        if (improvement_rate < -0.05) { // 拒绝大于5%的恶化
            return false;
        }

        // 3. 分层稳定性控制
        if (iteration < 10) { // 早期迭代
            return true; // 允许小幅恶化
        } else if (iteration < 35) { // 中期迭代
            // 停滞时允许小幅恶化
            return stagnation_count < max_stagnation;
        } else { // 后期迭代
            // 除非有非常好的理由，否则拒绝恶化
            return improvement_rate > -0.005; // 后期只允许非常小的恶化
        }
    }
    
    // 检查是否收敛
    void check_convergence() {
        if (fitness_history.size() < 5) return;
        
        // 方法1：检查适应度变化是否足够小
        double max_fitness = *max_element(fitness_history.begin(), fitness_history.end());
        double min_fitness = *min_element(fitness_history.begin(), fitness_history.end());
        double relative_range = (max_fitness - min_fitness) / max(abs(max_fitness), abs(min_fitness));
        
        if (relative_range < convergence_threshold) {
            convergence_detected = true;
        }
        
        // 方法2：检查连续小改进
        if (consecutive_small_improvements >= max_consecutive_small_improvements) {
            convergence_detected = true;
        }
    }
    
    // 获取当前最佳解决方案
    double* get_best_solution() { return historical_best_solution; }
    double get_best_fitness() { return historical_best_fitness; }
    bool is_converged() { return convergence_detected; }
    int get_stagnation_count() { return stagnation_count; }
    
    // 重置收敛状态（用于通信后的重新评估）
    void reset_convergence_if_improved(double new_fitness) {
        if (new_fitness < historical_best_fitness * 0.99) { // 显著改进
            convergence_detected = false;
            consecutive_small_improvements = 0;
        }
    }
};

// 获取当前时间（毫秒）
long getCurrentTime() {
    struct timeval tv;
    gettimeofday(&tv, NULL);
    return tv.tv_sec * 1000 + tv.tv_usec / 1000;
}

// 全局平均函数
double global_average(double value, vector<int> neighbor_id) {
    MPI_Status stat;
    double gval = value;
    int nei_num = neighbor_id.size();
    int rounds = 20;
    
    for (int i = 0; i < rounds; i++) {
        MPI_Request req[neighbor_id.size()];
        for (int rank_index = 0; rank_index < nei_num; rank_index++) {
            MPI_Isend(&gval, 1, MPI_DOUBLE, neighbor_id[rank_index], 0, MPI_COMM_WORLD, &req[rank_index]);
        }
        
        double nval_sum = 0;
        for (int rank_index = 0; rank_index < nei_num; rank_index++) {
            double nval;
            MPI_Recv(&nval, 1, MPI_DOUBLE, neighbor_id[rank_index], 0, MPI_COMM_WORLD, &stat);
            nval_sum += nval;
        }
        gval = (gval + nval_sum) / (nei_num + 1);
    }
    return gval;
}

int main(int argc, char* argv[]) {
    long start_time = getCurrentTime();
    
    // MPI初始化
    int myrank, nprocs;
    MPI_Init(&argc, &argv);
    MPI_Comm_size(MPI_COMM_WORLD, &nprocs);
    MPI_Comm_rank(MPI_COMM_WORLD, &myrank);
    MPI_Status stat;
    
    // 算法参数
    double distrub = 0.1;
    double gen_per_d = 0.4;
    int eva_per_d = 3000;
    string exID = "ex01";
    string method = "RL-MACPO-CSO-CONVERGENCE-STABLE"; 
    string funcID = argv[argc - 1];
    string outDir = "./output/3000/CSO/";
    
    double initial_penalty = 1.0;
    int swarm_size = 300;
    double prev_global_fit = 1e10;
    double ema_fitness = 1e6;
    
    srand(getCurrentTime());
    
    string filename = outDir + exID + "/" + exID + "_" + method + "_" + funcID + "_0.txt";
    
    // 创建基准函数和组件
    Benchmarks *pFunc = new Benchmarks(funcID, 0, true);
    int dimension = pFunc->getDimension();
    vector<int> groupDim = pFunc->getGroupDim(myrank);
    pFunc->max_eva_times = eva_per_d * groupDim.size();
    int gen_times = groupDim.size() * gen_per_d;
    
    // 初始化数组
    double *globalBest = new double[dimension];
    int* variable_switch = new int[dimension];
    int* compute_result = new int[dimension];
    memset(globalBest, 0, sizeof(double) * dimension);
    memset(variable_switch, 0, sizeof(int) * dimension);
    memset(compute_result, 0, sizeof(int) * dimension);
    
    // 创建收敛稳定性管理器
    ConvergenceStabilityManager* stability_manager = new ConvergenceStabilityManager(dimension, 5);
    
    // 创建组件
    evaluator *Evaluator = new EnhancedRLPenaltyEvaluator(variable_switch, compute_result, dimension, globalBest, initial_penalty, pFunc, myrank);
    optimizer *Optimizer = new optimizer_CSO(swarm_size, Evaluator, groupDim);
    competition *Competition = new EnhancedCompetition(pFunc, &(Optimizer->swarm), variable_switch);
    sharing *Sharing = new sharing_variable_wise(pFunc, ((competition_variable_wise*)Competition)->compete_result);
    
    Optimizer->init();
    
    // 获取重叠信息
    vector<int> overlapDim;
    vector<vector<int>> overlapDimForEach;
    vector<int> overlap_groups = pFunc->getOverlapGroup(myrank);
    for (int i : overlap_groups) {
        vector<int> overlap = pFunc->getOverlapDim(myrank, i);
        overlapDim.insert(overlapDim.end(), overlap.begin(), overlap.end());
        overlapDimForEach.push_back(overlap);
    }
    
    vector<int> commu_object = pFunc->getOverlapGroup(myrank);
    
    int iter = 0;
    int rejected_updates = 0;
    
    // 主循环
    while (!pFunc->reachMaxEva()) {
        // 1. 粒子群优化
        int success = 0;
        for (int gen = 0; gen < gen_times; gen++) {
            Optimizer->generation(success);
            if (pFunc->reachMaxEva()) break;
        }
        
        // 2. 获取当前最优解
        double *localBestPar = Optimizer->getBestPar();
        
        // 3. 计算冲突和通信决策
        double conflict_now = ((EnhancedRLPenaltyEvaluator*)Evaluator)->calculate_enhanced_conflict(localBestPar);
        bool should_communicate = ((EnhancedRLPenaltyEvaluator*)Evaluator)->should_communicate(localBestPar, iter);
        
        // 4. 全局通信同步
        int communicate_int = should_communicate ? 1 : 0;
        int global_communicate_int = 0;
        MPI_Allreduce(&communicate_int, &global_communicate_int, 1, MPI_INT, MPI_MAX, MPI_COMM_WORLD);
        should_communicate = (global_communicate_int > 0);
        
        // 5. 简化的协商过程（为节省代码长度）
        if (should_communicate) {
            // 基本通信逻辑（简化版本）
            memcpy(globalBest, localBestPar, dimension * sizeof(double));
        } else {
            memcpy(globalBest, localBestPar, dimension * sizeof(double));
        }
        
        // 6. 更新评估器和重新评估
        ((EnhancedRLPenaltyEvaluator*)Evaluator)->set_global_best(globalBest);
        Evaluator->total_evaluate(Optimizer->swarm);
        sort(Optimizer->swarm.begin(), Optimizer->swarm.end(), cmp_unit_pointer);
        ((optimizer_CSO*)Optimizer)->bestFit = Optimizer->swarm[0]->fitness;
        
        // 7. 计算全局适应度
        double local_fit = Evaluator->evaluate(globalBest);
        MPI_Barrier(MPI_COMM_WORLD);
        double local_sum = local_fit;
        double global_sum;
        MPI_Allreduce(&local_sum, &global_sum, 1, MPI_DOUBLE, MPI_SUM, MPI_COMM_WORLD);
        double global_fit = global_sum / nprocs;
        
        // 8. *** 关键改进：使用收敛稳定性管理器决定是否接受更新 ***
        if (iter > 0) {
            static double historical_best = 1e10;
            static bool ema_initialized = false;
            
            bool should_accept = stability_manager->should_accept_fitness(prev_global_fit, global_fit, globalBest, iter);
            
            historical_best = min(historical_best, global_fit);
            
            if (!ema_initialized) {
                ema_fitness = global_fit;
                ema_initialized = true;
            }
            
            ema_fitness = 0.8 * ema_fitness + 0.2 * global_fit;
            
            double improvement_rate = 0.0;
            if (abs(prev_global_fit) > 1e-8) {
                improvement_rate = (prev_global_fit - global_fit) / abs(prev_global_fit);
            }
            
            double reward = 0.0;
            if (global_fit > 0) {
                reward = -log(global_fit);
            } else {
                reward = -100.0;
            }
            
            ((EnhancedRLPenaltyEvaluator*)Evaluator)->update_rl_weight(localBestPar, reward);
            
            if (myrank == 0) {
                ofstream outfile(filename, ios::app);
                if (outfile.is_open()) {
                    outfile << scientific << uppercase << setprecision(2);
                    outfile << "Iter " << iter << ": Old fit=" << prev_global_fit 
                           << ", New fit=" << global_fit 
                           << ", Improvement=" << improvement_rate 
                           << ", Reward=" << reward 
                           << ", Conflict=" << conflict_now 
                           << ", EMA=" << ema_fitness 
                           << ", Historical_best=" << historical_best 
                           << ", Weight=" << ((EnhancedRLPenaltyEvaluator*)Evaluator)->get_alpha() 
                           << ", Converged=" << stability_manager->is_converged()
                           << ", Stagnation=" << stability_manager->get_stagnation_count() << endl;
                    outfile.close();
                }
            }
            
            // *** 根据稳定性管理器的决策接受或拒绝更新 ***
            if (should_accept) {
                prev_global_fit = global_fit;
                if (myrank == 0) {
                    ofstream outfile(filename, ios::app);
                    if (outfile.is_open()) {
                        outfile << "Iter " << iter << ": STABILITY ACCEPT improvement=" << improvement_rate << endl;
                        outfile.close();
                    }
                }
                // 重置收敛状态如果有显著改进
                stability_manager->reset_convergence_if_improved(global_fit);
            } else {
                rejected_updates++;
                if (myrank == 0) {
                    ofstream outfile(filename, ios::app);
                    if (outfile.is_open()) {
                        outfile << "Iter " << iter << ": STABILITY REJECT improvement=" << improvement_rate 
                               << " (Total rejected: " << rejected_updates << ")" << endl;
                        outfile.close();
                    }
                }
                // 不更新prev_global_fit，保持原值
            }
            
        } else {
            prev_global_fit = global_fit;
        }
        
        // 9. 早停检查：如果已收敛且停滞时间过长，考虑提前结束
        if (stability_manager->is_converged() && stability_manager->get_stagnation_count() > 10) {
            if (myrank == 0) {
                ofstream outfile(filename, ios::app);
                if (outfile.is_open()) {
                    outfile << "Iter " << iter << ": EARLY STOP due to convergence and stagnation" << endl;
                    outfile.close();
                }
            }
            // 可以选择在这里提前结束，但先继续运行看效果
        }
        
        // 10. 检查终止条件
        MPI_Barrier(MPI_COMM_WORLD);
        double reach_max_eva = pFunc->reachMaxEva() ? 1.0 : 0.0;
        double if_any_reach_max_eva = global_average(reach_max_eva, commu_object);
        if (if_any_reach_max_eva > 0) break;
        
        // 11. 记录进度
        if (myrank == 0) {
            ofstream outfile(filename, ios::app);
            if (outfile.is_open()) {
                outfile << scientific << uppercase << setprecision(2);
                outfile << pFunc->eva_count << ", " << prev_global_fit // 注意这里使用prev_global_fit（稳定版本）
                       << ", " << ((EnhancedRLPenaltyEvaluator*)Evaluator)->get_alpha() 
                       << ", " << conflict_now << endl;
                outfile.close();
            }
        }
        
        MPI_Barrier(MPI_COMM_WORLD);
        iter++;
    }
    
    // 最终输出
    MPI_Barrier(MPI_COMM_WORLD);
    long end_time = getCurrentTime();
    long total_time = end_time - start_time;
    
    if (myrank == 0) {
        std::string dir = outDir + exID + "/";
        system(("mkdir -p " + dir).c_str());
        
        ofstream outfile(filename, ios::app);
        if (outfile.is_open()) {
            outfile << scientific << uppercase << setprecision(2);
            outfile << pFunc->eva_count << ", " << prev_global_fit 
                   << ", " << total_time << "ms" << endl;
            outfile << "总评估次数: " << pFunc->eva_count 
                   << ", 最终适应度: " << prev_global_fit 
                   << ", 总运行时间: " << total_time << "ms" 
                   << ", 被拒绝更新数: " << rejected_updates 
                   << ", 历史最佳: " << stability_manager->get_best_fitness() << endl;
            outfile.close();
        }
        
        cout << "Final result: iterations=" << iter 
             << ", evals=" << pFunc->eva_count 
             << ", final fitness=" << prev_global_fit 
             << ", total time=" << total_time << "ms"
             << ", rejected updates=" << rejected_updates 
             << ", historical best=" << stability_manager->get_best_fitness() << endl;
             
        // 历史最佳回退机制
        if (prev_global_fit > stability_manager->get_best_fitness() * 1.05) { // 如果最终结果比历史最佳差5%以上
            cout << "最终结果比历史最佳差，回退到历史最佳..." << endl;
            prev_global_fit = stability_manager->get_best_fitness();
            cout << "回退后最终适应度: " << prev_global_fit << endl;
        }
    }
    
    // 清理资源
    delete[] globalBest;
    delete[] variable_switch;
    delete[] compute_result;
    delete stability_manager;
    delete Evaluator;
    delete Optimizer;
    delete Competition;
    delete Sharing;
    delete pFunc;
    
    MPI_Finalize();
    return 0;
}
